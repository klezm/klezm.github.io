import { Card, LinkCard } from '@astrojs/starlight/components';
import { getCollection } from 'astro:content';

export const blogPosts = (await getCollection('docs', ({ id }) => id.startsWith('blog/')))
  .sort((a, b) => new Date(b.data.date).valueOf() - new Date(a.data.date).valueOf());

## Posts

{blogPosts.map((post) => (
    <Card title={post.data.title}>
        {post.data.excerpt ?? ''}
        <br/>
        <a href={`/${post.id}/`} class="no-underline">Read more →</a>
    </Card>
))}
