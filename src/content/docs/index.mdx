import BlogPostCard from '../../components/BlogPostCard.astro';
import { getCollection } from 'astro:content';

export const blogPosts = (await getCollection('docs', ({ id }) => id.startsWith('blog/')))
  .sort((a, b) => new Date(b.data.date).valueOf() - new Date(a.data.date).valueOf());

<div class="mx-auto max-w-4xl px-4 py-8">
  {/* Hero Section */}
  <header class="mb-12 text-center">
    <h1 class="mb-4 bg-gradient-to-r from-purple-600 via-blue-500 to-cyan-500 bg-clip-text text-4xl font-bold text-transparent sm:text-5xl">
      klezm's blog
    </h1>
    <p class="text-lg text-gray-600 dark:text-gray-300">
      Computer science explorations, programming insights, and tech experiences.
    </p>
  </header>

  {/* Blog Posts Grid */}
  <section>
    <h2 class="mb-6 text-2xl font-bold text-gray-900 dark:text-white">
      Latest Posts
    </h2>
    <div class="grid gap-6 sm:grid-cols-1 lg:grid-cols-2">
      {blogPosts.map((post) => (
        <BlogPostCard
          title={post.data.title}
          date={post.data.date}
          excerpt={post.data.excerpt}
          href={`/${post.id}/`}
          tags={post.data.tags}
        />
      ))}
    </div>
  </section>
</div>
