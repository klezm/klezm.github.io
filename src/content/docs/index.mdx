import { getCollection } from 'astro:content';

export const blogPosts = (await getCollection('docs', ({ id }) => id.startsWith('blog/')))
  .sort((a, b) => new Date(b.data.date).valueOf() - new Date(a.data.date).valueOf());

export const formatDate = (date) => {
  return new Date(date).toLocaleDateString('en-US', {
    year: 'numeric',
    month: 'long',
    day: 'numeric'
  });
};

<div className="hero-section">
  <h1 className="hero-title">Welcome to My Blog</h1>
  <p className="hero-description">
    Exploring ideas in computer science, functional programming, and technology.
    Join me on this journey of learning and discovery.
  </p>
</div>

<h2 className="text-2xl font-bold mb-6 flex items-center gap-3">
  <span className="inline-block w-1 h-8 bg-[var(--sl-color-accent)] rounded-full"></span>
  Latest Posts
</h2>

<div className="blog-grid">
  {blogPosts.map((post, index) => (
    <article 
      key={post.id}
      className="blog-card animate-fade-in-up" 
      style={{ animationDelay: `${index * 0.1}s` }}
    >
      <div className="flex flex-col h-full">
        {post.data.tags && post.data.tags.length > 0 && (
          <div className="flex flex-wrap gap-2 mb-3">
            {post.data.tags.slice(0, 3).map((tag) => (
              <span key={tag} className="blog-tag">{tag}</span>
            ))}
          </div>
        )}
        <h3 className="text-xl font-bold mb-2 text-[var(--sl-color-white)]">
          <a href={`/${post.id}/`} className="no-underline hover:text-[var(--sl-color-accent)] transition-colors">
            {post.data.title}
          </a>
        </h3>
        {post.data.excerpt && (
          <p className="text-[var(--sl-color-gray-2)] mb-4 flex-grow line-clamp-3">
            {post.data.excerpt}
          </p>
        )}
        <div className="flex items-center justify-between mt-auto pt-4 border-t border-[var(--sl-color-gray-5)]">
          <div className="blog-meta">
            {post.data.date && (
              <span className="blog-meta-item">
                <svg xmlns="http://www.w3.org/2000/svg" className="w-4 h-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                  <rect x="3" y="4" width="18" height="18" rx="2" ry="2"></rect>
                  <line x1="16" y1="2" x2="16" y2="6"></line>
                  <line x1="8" y1="2" x2="8" y2="6"></line>
                  <line x1="3" y1="10" x2="21" y2="10"></line>
                </svg>
                {formatDate(post.data.date)}
              </span>
            )}
          </div>
          <a href={`/${post.id}/`} className="read-more-link no-underline">
            Read more
            <svg xmlns="http://www.w3.org/2000/svg" className="w-4 h-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
              <line x1="5" y1="12" x2="19" y2="12"></line>
              <polyline points="12 5 19 12 12 19"></polyline>
            </svg>
          </a>
        </div>
      </div>
    </article>
  ))}
</div>
